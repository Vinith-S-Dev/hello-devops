/* Perfect Number*/

for num in range(1,10001):
  sum=0
  for i in range(1,num):
    if num%i==0:
      sum +=i
      if num==sum:
        print(num)

/* Array*/

arr = [12,45,7,89,23]
max = arr[0]
sMax = arr[0]
for i in range(len(arr)):
  if arr[i]>max:
    sMax = max
    max = arr[i]
print(sMax)

/* Class and Objects */

class Book:
  library="City library"
  def __init__(self,title,author):
    self.title=title
    self.author=author
    self.available=True
  def borrow(self):
    self.available=False
  def __str__(self):
    return f"{self.title} by {self.author}"
